async function processAI() { const input = document.getElementById('userInput').value; const systemPrompt = "你是一位擅長 SEL 與正向管教的國小導師，請將以下描述轉化為親師溝通訊息："; const response = await fetch('/api/assistant', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ prompt: systemPrompt + input }) }); const result = await response.json(); document.getElementById('result').innerText = result.candidates[0].content.parts[0].text; }
